Google Sheets Connection

Store & load schedules directly in your Google Sheet

Paste your deployed Google Apps Script URL above to enable 2-way saving and loading.

1-Minute Google Sheet Setup
  1. Open a Google Sheet and go to Extensions > Apps Script.
  2. Paste the code below (replace any older version), click Deploy > New deployment > Web app.
  3. Set "Who has access" to Anyone, authorize, and paste the URL above.
var CHUNK = 40000; // Sheets cells hold max 50,000 characters

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse(e.postData.contents);
    var json = JSON.stringify(payload);

    // 1. Full app state, split across rows so it never hits the cell limit
    var stateSheet = ss.getSheetByName("AppState") || ss.insertSheet("AppState");
    stateSheet.clear();
    var rows = [];
    for (var i = 0; i < json.length; i += CHUNK) {
      rows.push([json.substring(i, i + CHUNK)]);
    }
    stateSheet.getRange(1, 1, rows.length, 1).setValues(rows);

    // 2. Human-readable summary: one row per cat, rewritten on each sync
    var logSheet = ss.getSheetByName("CareLog") || ss.insertSheet("CareLog");
    logSheet.clear();
    logSheet.appendRow(["Last synced", "Cat Name", "Active Items", "Latest Note"]);
    var patients = payload.patients || [];
    patients.forEach(function(p) {
      var items = (p.regimen || []).map(function(r) {
        return r.name + " (every " + r.intervalHours + "h, offset " + r.offsetMinutes + "m)";
      }).join("; ");
      var note = (p.notes && p.notes.length) ? p.notes[0].text : "";
      logSheet.appendRow([new Date(), p.name, items, note]);
    });

    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: String(err)}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var stateSheet = ss.getSheetByName("AppState");
  var data = "{}";
  if (stateSheet && stateSheet.getLastRow() > 0) {
    data = stateSheet.getRange(1, 1, stateSheet.getLastRow(), 1).getValues()
      .map(function(r) { return r[0]; }).join("");
  }
  return ContentService.createTextOutput(data)
    .setMimeType(ContentService.MimeType.JSON);
}
